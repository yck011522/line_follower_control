#include "NfcAsync.h"
#include <utility>

namespace {
constexpr uint8_t kAddress = 0x28;
}

// The datasheet allows N bytes at one register address, including FIFO burst writes.
bool NfcAsync::write(uint8_t reg, const uint8_t *values, size_t length)
{
    Wire.beginTransmission(kAddress);
    Wire.write(reg);
    Wire.write(values, length);
    ++transactions_;
    if (Wire.endTransmission(true)) { pointer_ = 0xFF; return false; }
    pointer_ = reg;
    return true;
}

// Single-byte register changes share the same error handling as burst writes.
bool NfcAsync::writeByte(uint8_t reg, uint8_t value)
{
    return write(reg, &value, 1);
}

// A pointer write is a separate step, keeping every service call to one Wire transaction.
// Reads do not increment the FM17550 register address (datasheet section 4.4.7).
NfcAsync::ReadStep NfcAsync::readStep(uint8_t reg, uint8_t *values, size_t length)
{
    if (pointer_ != reg) {
        Wire.beginTransmission(kAddress);
        Wire.write(reg);
        // ESP32 Wire defers a no-STOP write and combines it with the following requestFrom.
        // This saves driver/bus overhead, at the cost of a slightly longer individual service call.
        if (!options_.combined) ++transactions_;
        if (Wire.endTransmission(!options_.combined)) { pointer_ = 0xFF; return ReadStep::Failed; }
        pointer_ = reg;
        if (!options_.combined) return ReadStep::Pending;
    }
    ++transactions_;
    if (Wire.requestFrom(kAddress, length, true) != length || Wire.available() != int(length)) {
        while (Wire.available()) Wire.read();
        pointer_ = 0xFF;
        return ReadStep::Failed;
    }
    for (size_t i = 0; i < length; ++i) values[i] = static_cast<uint8_t>(Wire.read());
    if (!options_.cachePointer) pointer_ = 0xFF;
    return ReadStep::Done;
}

// Only one-time initialization uses a blocking pointer/read pair; RF waits never do.
bool NfcAsync::readBlocking(uint8_t reg, uint8_t &value)
{
    ReadStep step = readStep(reg, &value, 1);
    if (step == ReadStep::Pending) step = readStep(reg, &value, 1);
    return step == ReadStep::Done;
}

// Checking static settings catches the previously observed startup/write-retention failure.
bool NfcAsync::configure(uint8_t reg, uint8_t value)
{
    uint8_t actual = 0;
    return writeByte(reg, value) && readBlocking(reg, actual) && actual == value;
}

// Reset/configure once per session, with RF off until all settings acknowledge/read back.
bool NfcAsync::begin(const Options &options)
{
    if (busy() || options.clockHz < 100000 || options.clockHz > 400000 ||
        options.rfTimeoutUs < 500 || options.rfTimeoutUs > 25000 || options.pollUs > 5000 ||
        options.gain < 4 || options.gain > 7) return false;
    options_ = options;
    initialized_ = false;
    readyTag_ = false;
    state_ = State::Idle;
    result_ = Result::None;
    pointer_ = 0xFF;
    commandActive_ = fifoEmpty_ = false;
    framing_ = 0xFF;
    if (!Wire.setClock(options.clockHz)) return false;
    Wire.setTimeOut(2); // Bounds each synchronous transfer under a bus fault.
    uint8_t version = 0, command = 0, gain = 0, demod = 0;
    if (!readBlocking(0x37, version) || !writeByte(0x14, 0x80) || !writeByte(0x01, 0x0F)) return false;
    delay(10);
    const uint32_t start = millis();
    do {
        if (!readBlocking(0x01, command)) return false;
        if (!(command & 0x1F)) break;
        delay(1);
    } while (millis() - start < 100);
    if (command & 0x1F) return false;
    if (!readBlocking(0x26, gain) || !readBlocking(0x19, demod) ||
        !configure(0x26, (gain & 0x8F) | (options.gain << 4)) || !configure(0x19, demod & 0x1F)) return false;
    // A 25 us timer tick is generated by prescaler 169. Round timeout up to a tick.
    const uint16_t reload = (options.rfTimeoutUs + 24) / 25;
    for (const auto &setting : {std::pair<uint8_t, uint8_t>{0x18, 0x55}, {0x12, 0x00},
             {0x13, 0x00}, {0x15, 0x40}, {0x16, 0x10}, {0x17, 0x80}, {0x11, 0x3D},
             {0x2A, 0x80}, {0x2B, 0xA9}, {0x2C, uint8_t(reload >> 8)}, {0x2D, uint8_t(reload)}}) {
        if (!configure(setting.first, setting.second)) return false;
    }
    if (!writeByte(0x08, 0x00) || !writeByte(0x0E, 0x00) || !configure(0x14, 0x83)) return false;
    delay(10); // Cold-tag power-up belongs to initialization, not steady-state read latency.
    Serial.printf("INIT version=0x%02X clock=%lu timeout_us=%u\n", version,
                  static_cast<unsigned long>(options.clockHz), options.rfTimeoutUs);
    transactions_ = 0;
    emptyFrames_ = 0;
    retryAfterAbort_ = false;
    initialized_ = true;
    return true;
}

// A remembered READY state chooses a faster command, but every UID still comes from fresh RF bytes.
bool NfcAsync::startRead()
{
    if (!initialized_ || busy() || result_ != Result::None) return false;
    emptyRetries_ = 0;
    frame(!options_.adaptive || !readyTag_);
    return true;
}

// Prepare local state only; the next service call begins the actual bus work.
void NfcAsync::frame(bool discovery)
{
    discovery_ = discovery;
    irq_ = error_ = count_ = control_ = loadIndex_ = dataIndex_ = 0;
    state_ = options_.fastTx && commandActive_ ? State::Flush : State::Stop;
}

// A bus fault invalidates initialization; protocol/no-tag events can be retried normally.
void NfcAsync::fail(Result result)
{
    if (result != Result::ProtocolError) readyTag_ = false;
    result_ = result;
    if (result == Result::BusError) initialized_ = false;
    state_ = State::Abort;
}

// Discovery supplies ATQA; anticollision supplies four UID bytes plus their XOR checksum.
// SELECT is deliberately omitted because it is unnecessary for UID-only identification.
void NfcAsync::finishFrame()
{
    if (discovery_) {
        if (count_ != 2 || (rx_[0] & 0xC0)) { fail(Result::ProtocolError); return; }
        frame(false);
        return;
    }
    if (count_ != 5 || rx_[0] == 0x88 || (rx_[0] ^ rx_[1] ^ rx_[2] ^ rx_[3]) != rx_[4]) {
        fail(Result::ProtocolError);
        return;
    }
    memcpy(uid_, rx_, 4);
    readyTag_ = true;
    result_ = Result::Uid;
    state_ = State::Idle;
}

// Advance a single operation. RF reception is delegated to the chip while the caller runs other work.
void NfcAsync::service(uint64_t nowUs)
{
    if (state_ == State::Idle) return;
    const uint8_t tx[] = {discovery_ ? uint8_t(0x52) : uint8_t(0x93), uint8_t(0x20)};
    const uint8_t bits = discovery_ ? 7 : 0;
    bool ok = true;
    ReadStep step;
    switch (state_) {
    case State::Stop:
        ok = writeByte(0x01, 0x00); commandActive_ = false; state_ = State::Flush; break;
    // Clear completion flags after preparing FIFO/framing so old flags cannot survive preparation.
    case State::Clear:
        ok = writeByte(0x04, 0x7F);
        state_ = options_.fastTx && commandActive_ ? State::Send : State::Command;
        break;
    case State::Flush:
        if (!(options_.fastTx && fifoEmpty_)) ok = writeByte(0x0A, 0x80);
        fifoEmpty_ = true; state_ = State::Framing; break;
    case State::Framing:
        // Rewriting StartSend=1 triggers the next transmission; stable framing needs no extra write.
        if (!(options_.fastTx && framing_ == bits)) ok = writeByte(0x0D, bits);
        framing_ = bits; state_ = State::Load; break;
    case State::Load:
        fifoEmpty_ = false;
        if (options_.burst) {
            ok = write(0x09, tx, discovery_ ? 1 : 2);
            state_ = State::Clear;
        } else {
            ok = writeByte(0x09, tx[loadIndex_++]);
            if (loadIndex_ == (discovery_ ? 1 : 2)) state_ = State::Clear;
        }
        break;
    case State::Command:
        // The datasheet keeps Transceive active until another command cancels it.
        ok = writeByte(0x01, 0x0C); commandActive_ = true; state_ = State::Send; break;
    case State::Send:
        ok = writeByte(0x0D, 0x80 | bits);
        nextPollUs_ = nowUs + options_.pollUs;
        deadlineUs_ = nowUs + options_.rfTimeoutUs + 2000;
        state_ = State::Poll;
        break;
    case State::Poll:
        if (nowUs < nextPollUs_) return;
        step = readStep(0x04, &irq_, 1);
        if (step == ReadStep::Failed) { ok = false; break; }
        if (step == ReadStep::Done) {
            nextPollUs_ = nowUs + options_.pollUs;
            if (irq_ & 0x20) state_ = State::Error;
            else if ((irq_ & 0x01) || nowUs >= deadlineUs_) {
                // A tag introduced into the field may be IDLE. Rediscover after a failed READY probe.
                if (!discovery_ && options_.adaptive && readyTag_) { readyTag_ = false; frame(true); }
                else fail(Result::NoTag);
            }
            else if (irq_ & 0x02) state_ = State::Error;
        }
        break;
    case State::Error:
        step = readStep(0x06, &error_, 1);
        if (step == ReadStep::Failed) ok = false;
        else if (step == ReadStep::Done) {
            if (error_) fail(Result::ProtocolError);
            else state_ = State::Count;
        }
        break;
    case State::Count:
        step = readStep(0x0A, &count_, 1);
        if (step == ReadStep::Failed) ok = false;
        else if (step == ReadStep::Done) {
            count_ &= 0x7F;
            // An empty RxIRq is rejected, never published as a UID. Retry the same RF command once.
            if (!count_ && emptyRetries_++ == 0) {
                ++emptyFrames_;
                retryAfterAbort_ = true;
                state_ = State::Abort;
            }
            else if (count_ != (discovery_ ? 2 : 5)) {
                if (!count_) ++emptyFrames_;
                fail(Result::ProtocolError);
            }
            else state_ = State::Control;
        }
        break;
    case State::Control:
        step = readStep(0x0C, &control_, 1);
        if (step == ReadStep::Failed) ok = false;
        else if (step == ReadStep::Done) {
            if (control_ & 7) fail(Result::ProtocolError);
            else state_ = State::Data;
        }
        break;
    case State::Data:
        step = readStep(0x09, rx_ + dataIndex_, options_.burst ? count_ : 1);
        if (step == ReadStep::Failed) ok = false;
        else if (step == ReadStep::Done) {
            dataIndex_ += options_.burst ? count_ : 1;
            if (dataIndex_ == count_) { fifoEmpty_ = true; finishFrame(); }
        }
        break;
    case State::Abort:
        // This write terminates a timed-out command before exposing the event to the application.
        if (!writeByte(0x01, 0x00)) { result_ = Result::BusError; initialized_ = false; retryAfterAbort_ = false; }
        commandActive_ = fifoEmpty_ = false;
        framing_ = 0xFF;
        if (retryAfterAbort_) { retryAfterAbort_ = false; frame(discovery_); }
        else state_ = State::Idle;
        break;
    case State::Idle: break;
    }
    if (!ok) fail(Result::BusError);
}

// An event remains latched until consumed; an old UID is not valid for timeout/error events.
NfcAsync::Result NfcAsync::takeResult(uint8_t uid[4])
{
    if (busy()) return Result::None;
    const Result value = result_;
    result_ = Result::None;
    if (value == Result::Uid) memcpy(uid, uid_, 4);
    return value;
}

// Session cleanup always attempts both writes, even if the first one fails.
bool NfcAsync::end()
{
    state_ = State::Idle;
    initialized_ = readyTag_ = false;
    result_ = Result::None;
    const bool idle = writeByte(0x01, 0x00);
    const bool off = writeByte(0x14, 0x80);
    uint8_t value = 0;
    const bool verified = readBlocking(0x14, value) && !(value & 3);
    return idle && off && verified;
}
